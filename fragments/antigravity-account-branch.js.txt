			/* T3_ANTIGRAVITY_HUB_USAGE_PATCH_V1 */
			if (account.provider === "antigravity") {
				const endpoints = [
					"https://daily-cloudcode-pa.googleapis.com",
					"https://cloudcode-pa.googleapis.com"
				];
				const windowsById = new Map();
				const addBucket = (bucket, groupName, requestedAt) => {
					if (bucket.disabled || !bucket.bucketId || typeof bucket.remainingFraction !== "number" || !Number.isFinite(bucket.remainingFraction)) return;
					const weekly = bucket.window === "weekly";
					const session = bucket.window === "5h";
					const windowDurationMins = weekly ? 7 * 24 * 60 : session ? 5 * 60 : void 0;
					const untouched = bucket.remainingFraction === 1 && bucket.description === void 0 && windowDurationMins !== void 0 && Math.abs(Date.parse(bucket.resetTime ?? "") - requestedAt - windowDurationMins * 60_000) <= 60_000;
					const family = bucket.bucketId.startsWith("gemini-argon-") ? groupName?.trim() || "Gemini Argon" : bucket.bucketId.startsWith("gemini-") ? "Gemini" : bucket.bucketId.startsWith("3p-") ? "Claude & GPT (shared)" : groupName?.trim() || bucket.displayName?.trim() || bucket.bucketId;
					const resetMs = untouched ? NaN : Date.parse(bucket.resetTime ?? "");
					const window = {
						id: bucket.bucketId,
						label: `${family} · ${weekly ? "Weekly" : session ? "5-hour" : bucket.displayName?.trim() || "Quota"}`,
						kind: weekly ? "weekly" : session ? "session" : "other",
						usedPercent: (1 - Math.max(0, Math.min(1, bucket.remainingFraction))) * 100,
						...Number.isFinite(resetMs) ? { resetsAt: new Date(resetMs).toISOString() } : {},
						...windowDurationMins === void 0 ? {} : { windowDurationMins }
					};
					const previous = windowsById.get(window.id);
					if (!previous || window.usedPercent > previous.usedPercent) windowsById.set(window.id, window);
				};
				for (const endpoint of endpoints) {
					const requestedAt = Date.now();
					const summary = yield* apiCall(hub, account, `${endpoint}/v1internal:retrieveUserQuotaSummary`, account.project_id ? { project: account.project_id } : {}).pipe(flatMap(decodeAntigravitySummary), orElseSucceed(() => void 0));
					if (!summary) continue;
					const groups = summary.groups?.some((group) => group.buckets?.length) ? summary.groups : [{ buckets: summary.buckets }];
					for (const group of groups ?? []) for (const bucket of group.buckets ?? []) addBucket(bucket, group.displayName, requestedAt);
				}
				const models = yield* apiCall(hub, account, "https://cloudcode-pa.googleapis.com/v1internal:fetchAvailableModels", account.project_id ? { project: account.project_id } : {}).pipe(flatMap(decodeAntigravityModels), orElseSucceed(() => void 0));
				for (const [modelId, model] of Object.entries(models?.models ?? {})) {
					const family = model.modelProvider === "MODEL_PROVIDER_GOOGLE" || modelId.startsWith("gemini-") ? "Gemini" : model.modelProvider === "MODEL_PROVIDER_ANTHROPIC" || modelId.startsWith("claude-") ? "Claude" : model.modelProvider === "MODEL_PROVIDER_OPENAI" || modelId.startsWith("gpt-") ? "GPT" : void 0;
					if (!family) continue;
					for (const [suffix, quota] of [["quota", model.quotaInfo], ["weekly", model.weeklyQuotaInfo]]) {
						if (!quota || typeof quota.remainingFraction !== "number" || !Number.isFinite(quota.remainingFraction)) continue;
						const alreadyCovered = [...windowsById.values()].some((window) => {
							if (window.kind !== (suffix === "weekly" ? "weekly" : "session")) return false;
							if (family === "Gemini") return window.id.startsWith("gemini-");
							return window.id.startsWith("3p-") || window.id.startsWith(`${family.toLowerCase()}-`);
						});
						if (alreadyCovered) continue;
						const window = {
							id: `${family.toLowerCase()}-${suffix}`,
							label: `${family} · ${suffix === "weekly" ? "Weekly" : "Reported quota"}`,
							kind: suffix === "weekly" ? "weekly" : "other",
							usedPercent: (1 - Math.max(0, Math.min(1, quota.remainingFraction))) * 100,
							...typeof quota.resetTime === "string" && Number.isFinite(Date.parse(quota.resetTime)) ? { resetsAt: new Date(Date.parse(quota.resetTime)).toISOString() } : {},
							...suffix === "weekly" ? { windowDurationMins: 7 * 24 * 60 } : {}
						};
						const previous = windowsById.get(window.id);
						if (!previous || window.usedPercent > previous.usedPercent) windowsById.set(window.id, window);
					}
				}
				return {
					...base,
					plan: "Antigravity Subscription",
					usageLimits: { checkedAt, windows: [...windowsById.values()] }
				};
			}
